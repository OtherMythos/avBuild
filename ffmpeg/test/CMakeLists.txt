cmake_minimum_required(VERSION 3.20)
project(avFFmpegTest LANGUAGES CXX)
find_package(AVFFmpeg CONFIG REQUIRED)
add_executable(avFFmpegTest main.cpp)
target_compile_features(avFFmpegTest PRIVATE cxx_std_17)
target_link_libraries(avFFmpegTest PRIVATE AVFFmpeg::FFmpeg)
add_library(avFFmpegShared SHARED shared.cpp)
target_compile_features(avFFmpegShared PRIVATE cxx_std_17)
target_link_libraries(avFFmpegShared PRIVATE AVFFmpeg::FFmpeg)
set_target_properties(avFFmpegShared PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
    set_property(TARGET avFFmpegTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    set_property(TARGET avFFmpegShared PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()
